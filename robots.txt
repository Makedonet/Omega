# Страница намеренно закрыта от индексации метатегом
# <meta name="robots" content="noindex,nofollow"> в index.html.
#
# Обход при этом разрешён сознательно: запрет через Disallow не даёт роботу

# без описания и без возможности его оттуда убрать.
